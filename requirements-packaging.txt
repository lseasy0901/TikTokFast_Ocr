# -*- coding: utf-8 -*-
# PP-OCRv6 生产打包环境 —— 构建机一次性安装（不随正式包发布）
#
# 用法（在项目 venv 里，构建机执行一次）：
#     venv\Scripts\python.exe -m pip install -r requirements-packaging.txt
#
# 与 requirements.txt 的分工
# --------------------------
# requirements.txt           应用运行基线（PySide6 / OpenCV / 更新客户端 …），
#                            源码环境与构建环境都要装。本文件**不重复**它的条目。
# requirements-packaging.txt 构建环境**额外**需要的：生产 OCR 后端的运行时，
#                            以及打包工具本身。
#
# 不再有 paddlepaddle-gpu / paddleocr / paddlex，也没有 nvidia-* CUDA wheel。
# OCR 后端已迁到 ONNX Runtime + DirectML（见 ocr/onnx_engine.py）：DirectML 是
# GPU 路径、CPUExecutionProvider 是回退，两者都编译在 onnxruntime 自身里，
# 不需要外部 CUDA 运行时，也不再有 paddle 专用索引。

# ---------------------------------------------------------------------------
# 1. 应用运行时依赖：ONNX Runtime（DirectML 版）
#
#    这个包会被 PyInstaller 打进正式包（连同 runtime/onnx/ 下的两个产物，
#    它们由 douyin_viewer.spec 的 datas 提供）。
#
#    为什么是 onnxruntime-directml 而不是 onnxruntime / onnxruntime-gpu：
#      * onnxruntime-gpu 需要外部 CUDA/cuDNN，正是本次要摆脱的依赖；
#      * 目标机是 Windows + DirectML，DirectML 版自带 DirectML.dll，
#        并且**同时**提供 CPUExecutionProvider 作为回退；
#      * onnxruntime 与 onnxruntime-directml 是同名互斥的两个发行包，
#        只能装一个。本文件钉的是 DirectML 版。
#
#    版本必须钉死，且与 Step 2 验收（107/107 parity、DirectML 实测）用的
#    是同一个版本：1.24.4。升级前必须重跑 scripts/verify_onnx_model.py 与
#    冻结自检，并按 scripts/onnx_model_manifest.json 重新核对产物。
# ---------------------------------------------------------------------------
onnxruntime-directml==1.24.4

# numpy 由应用与 ONNX Runtime 共用（ORT 用它做输入/输出的张量桥接），
# 所以在打包环境里也钉死，避免 pip 解析到不同版本。
numpy==2.3.5

# ---------------------------------------------------------------------------
# 2. 仅打包需要的工具（不随正式包发布）
#
#    pyinstaller-hooks-contrib 是必需的，不是可选：它自带
#    hook-onnxruntime.py（collect_dynamic_libs("onnxruntime")），
#    onnxruntime/capi 下的 onnxruntime.dll / DirectML.dll /
#    onnxruntime_providers_shared.dll 全靠它收集。缺了它，spec 不会报错，
#    只会打出一个 OCR 起不来的包。
#
#    velopack 不在这里：它是更新客户端的**运行**依赖，已经钉在 requirements.txt
#    里（velopack==1.2.0），避免同一版本号出现两个来源。
# ---------------------------------------------------------------------------
pyinstaller==6.22.2
pyinstaller-hooks-contrib==2026.7
